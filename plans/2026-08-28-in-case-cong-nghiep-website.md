# IN CASE CÔNG NGHIỆP Website Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use compose:subagent (recommended) or compose:execute to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-page static investor/partner website for IN CASE CÔNG NGHIỆP that communicates capital requirements, unit economics, utilization scenarios, payback timelines, and deployment costs — with an interactive ROI calculator.

**Architecture:** Vite vanilla TypeScript project. Semantic HTML page composed by modular TS section renderers. All charts are inline SVG or CSS. Finance calculations are pure functions tested with Vitest. No backend, no framework runtime, no external CDN. Built `dist/` deployed to Hostinger static hosting.

**Tech Stack:** Vite 6.x, TypeScript 5.x, Vitest, Playwright (smoke tests only), npm. No React, no Vue, no external charting library, no external fonts.

## Global Constraints

- All financial constants come from `src/data/business-model.ts` — never hardcode numbers in section renderers.
- All finance logic lives in `src/lib/finance.ts` as pure functions — no side effects, no DOM access.
- Colors use CSS custom properties from `src/styles/tokens.css` — never hardcode hex values in component CSS.
- Typography uses system font stacks defined in tokens — no Google Fonts, no `@font-face` with external URLs.
- Every `<section>` in HTML has an `id` matching the artifact structure (e.g., `id="hero"`, `id="investment-summary"`).
- Charts are inline `<svg>` elements generated by TypeScript — no `<canvas>`, no Chart.js, no D3.
- Calculator is browser-only: all computation in `src/components/roi-calculator.ts` using `finance.ts` functions.
- Print CSS in `src/styles/print.css` — hides interactive elements, ensures readable layout.
- Hostinger prices must include source URL `https://www.hostinger.com/vn/web-hosting` and access date `2026-08-28`.
- GitHub repo will be **private**. Do not assume GitHub Pages is free for private repos.
- No TBD, no TODO, no "similar to" — every step has exact code or exact command.

```typescript
export function formatVND(value: number): string {
  return new Intl.NumberFormat('vi-VN', {

    currency: 'VND',

  }).format(value);
}

export function formatPercent(value: number): string {
  return `${(value * 100).toFixed(0)}%`;
}

export function formatMonths(value: number): string {
  if (value < 1) return `${(value * 30).toFixed(0)} ngày`;
  return `${value.toFixed(1)} tháng`;
}

export function formatNumber(value: number): string {
  return new Intl.NumberFormat('vi-VN').format(value);
```

```typescript
/** Contribution margin per unit */
export function contributionMargin(sellPrice: number, directCost: number): number {
  return sellPrice - directCost;
}

/** Contribution per machine-hour */
export function contributionPerHour(contribution: number, hours: number): number {
  return contribution / hours;
}

/** Monthly capital recovery = (contribution/hour * sold hours) - fixed costs */
export function monthlyRecovery(
  contributionPerHourValue: number,
  soldHours: number,
  fixedCost: number,
): number {
  return contributionPerHourValue * soldHours - fixedCost;
}

/** Payback period in months */
export function paybackMonths(capital: number, monthlyRecoveryValue: number): number {
  if (monthlyRecoveryValue <= 0) return Infinity;
  return capital / monthlyRecoveryValue;
}

/** Maximum available machine hours per month */
export function maxHoursPerMonth(hoursPerDay: number, daysPerMonth: number): number {
  return hoursPerDay * daysPerMonth;
}
```

```typescript
        <p>Hoàn vốn: <span id="result-payback" class="data-value metric-highlight">${formatMonths(4.69)}</span></p>
      </div>
    </div>
  `;

  const utilizationInput = container.querySelector('#utilization-input') as HTMLInputElement;
  const contributionInput = container.querySelector('#contribution-input') as HTMLInputElement;
  const fixedCostInput = container.querySelector('#fixed-cost-input') as HTMLInputElement;

  const resultRecovery = container.querySelector('#result-recovery')!;
  const resultPayback = container.querySelector('#result-payback')!;

  function update(): void {
    const utilization = parseInt(utilizationInput.value, 10) / 100;
    const contribution = parseInt(contributionInput.value, 10);
    const fixedCost = parseInt(fixedCostInput.value, 10);
    const soldHours = Math.round(MAX_HOURS * utilization);

  contributionInput.addEventListener('input', update);
  fixedCostInput.addEventListener('input', update);
```

```typescript
import type { Scenario } from '../data/business-model';
import { formatVND, formatPercent } from '../lib/format';

export function createUtilizationChart(container: HTMLElement, scenarios: Scenario[]): void {
  const barHeight = 40;
  const gap = 16;
  const labelWidth = 120;
  const valueWidth = 160;
  const chartWidth = 600;
  const maxBarWidth = chartWidth - labelWidth - valueWidth;
  const maxRecovery = Math.max(...scenarios.map(s => s.monthlyRecovery));
  const totalHeight = scenarios.length * (barHeight + gap) - gap + 20;

  const bars = scenarios.map((s, i) => {
    const y = i * (barHeight + gap);
    const barWidth = Math.max(4, (s.monthlyRecovery / maxRecovery) * maxBarWidth);
    const fillColor = s.label === 'aggressive' ? 'var(--signal)' : 'var(--blueprint)';
    return `
      <g transform="translate(0, ${y})">
        <text x="${labelWidth - 8}" y="${barHeight / 2 + 5}" text-anchor="end"
              font-family="var(--font-heading)" font-size="14" fill="var(--ink)">
          ${s.name} ${formatPercent(s.utilization)}
        </text>
        <rect x="${labelWidth}" y="0" width="${barWidth}" height="${barHeight}"
              fill="${fillColor}" rx="2"/>
        <text x="${labelWidth + barWidth + 8}" y="${barHeight / 2 + 5}"
              font-family="var(--font-data)" font-size="13" fill="var(--ink)">
          ${formatVND(s.monthlyRecovery)}/tháng
        </text>
      </g>
    `;
  }).join('');

  container.innerHTML = `
    <svg viewBox="0 0 ${chartWidth} ${totalHeight}" role="img"
         aria-label="Biểu đồ thu hồi vốn theo kịch bản công suất"
         style="max-width: 100%; height: auto;">
      ${bars}
    </svg>
  `;
}
```
